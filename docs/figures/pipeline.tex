\begin{tikzpicture}[
  font=\small,
  >={Stealth[length=2.4mm]},

  tn/.style={circle, draw=enc!70, fill=enc!15, inner sep=0pt, minimum size=3.2mm},
  note/.style={font=\scriptsize\itshape, text=black!60},
  lbl/.style={rounded corners=2.5pt, draw=#1!65, fill=#1!12, line width=.6pt,

%==================== STAGE 1 : GRAPH ENCODING ====================
% curated hierarchy (mini tree)
\node[tn] (r)  at (0,0)       {};
\node[tn] (t1) at (-1,-0.85)  {};
\node[tn] (t2) at (1,-0.85)   {};
\node[tn] (l1) at (-1.5,-1.7) {};
\node[tn] (l2) at (-0.5,-1.7) {};
\node[tn] (l3) at (0.5,-1.7)  {};
\node[tn] (l4) at (1.5,-1.7)  {};
\draw[enc!55,line width=.7pt] (r)--(t1) (r)--(t2) (t1)--(l1) (t1)--(l2) (t2)--(l3) (t2)--(l4);
\node[fit=(r)(t1)(t2)(l1)(l4), inner sep=6pt] (Atree) {};
\node[note, above=3.5mm of Atree, text width=46mm, align=center] (Acap)
  {curated hierarchy\\ leaves = attributed distributions};

% encoder
\node[box=enc, below=15mm of Atree] (Aenc)
  {\textbf{Tree-GRU + GKI encoder}\\[1pt]\scriptsize M\"obius/Einstein midpoint\\\scriptsize depth-aware text injection};

% Poincare ball
\draw[enc!65,line width=1pt] (0,-7.4) circle (1.35);
\fill[enc!25] (0,-7.4) circle (0.07);
\foreach \a/\rr in {20/1.2,70/0.9,130/1.25,200/0.7,255/1.15,310/1.0,160/0.5,340/1.25,95/1.3}
  {\fill[enc!75!black] ($(0,-7.4)+(\a:\rr)$) circle (0.055);}
\node[circle,minimum size=2.9cm] (Adisk) at (0,-7.4) {};
\node[note, below=0pt of Adisk] {$h_{\text{all}}$ on Poincar\'e ball,\ \ $\rho=\sqrt{c}\,\lVert h\rVert$};

\draw[flow] (Atree) -- (Aenc);
\draw[flow] (Aenc)  -- (Adisk);

%==================== STAGE 2 : DIVERGENCE SCOUT ====================
\node[io] (Q) at (6,0.6) {\textbf{User question}};

\node[box=sct] (Banchor) at (6,-1.6) {\textbf{Anchor + relevance field}\\\scriptsize MiniLM cosine $\to$ rel$(v)$};
\node[box=sct] (Bcloud)  at (6,-3.7) {\textbf{Relevance-gated child clouds}\\\scriptsize $P_i,P_j$ per branch};
\node[box=sct] (Bwass)   at (6,-5.8) {\textbf{Wasserstein divergence}\\\scriptsize $W=\min_{\gamma}\langle\gamma,C\rangle$\\\scriptsize geodesic ground cost $C$};
\node[box=sct] (Bforks)  at (6,-7.9) {\textbf{Top-$k$ divergent forks}\\\scriptsize score $=\text{rel}^{\alpha}\!\cdot W$\\\scriptsize branch A vs B + drivers};

\draw[flow] (Q)       -- (Banchor);
\draw[flow] (Banchor) -- (Bcloud);
\draw[flow] (Bcloud)  -- (Bwass);
\draw[flow] (Bwass)   -- (Bforks);

% embeddings feed the scout (h_all store -> clouds)
\draw[dflow] (Adisk.east) -- (3,-7.4) |- (Bcloud.west);
\node[lbl=enc] at (3,-5.5) {$h_{\text{all}}$};   % dashed arrow passes through this box

%==================== STAGE 3 : REASONER ====================
\node[box=rsn] (Cctx) at (12,-1.3) {\textbf{Fork context}\\\scriptsize forks $\to$ distributions\\\scriptsize $+$ user question};
\node[box=rsn] (Cllm) at (12,-3.6) {\textbf{Frozen LLM (reasoner)}\\\scriptsize \texttt{<think>} triage relevance\\\scriptsize \texttt{<answer>} pluralistic};
\node[io]      (Cans) at (12,-5.9) {\textbf{Pluralistic answer}\\\scriptsize covers the range of views};

\draw[flow] (Bforks.east) -- (9,-7.9) |- (Cctx.west);
\node[lbl=sct] at (9,-4.6) {top-$k$ forks};   % arrow passes through this box
\draw[flow] (Cctx) -- (Cllm);
\draw[flow] (Cllm) -- (Cans);
